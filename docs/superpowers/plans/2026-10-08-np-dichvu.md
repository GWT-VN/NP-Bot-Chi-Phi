# Dịch vụ nền bot NP (`np-dichvu`) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bot NP ghi sổ chi phí từ nhóm Zalo chạy thành **một dịch vụ nền** (launchd) gọi `claude -p --bare` cho từng tin. Dịch vụ dùng SQLite, hàng đợi, hộp thư gửi và cảnh báo, thay cho bản channel/tmux trên nhánh `zalo`.

**Architecture:**
- 7 file trong `dichvu/`:
  - `csdl.mjs`: SQLite, bảng `tin`, luật hiểu trả lời, adapter cho so-core;
  - `zalo.mjs`: nhận tin + tải ảnh;
  - `hop-thu.mjs`: hộp thư gửi + cảnh báo;
  - `claude.mjs`: chạy `claude -p`;
  - `cong-cu.mjs`: MCP HTTP theo token việc;
  - `hang-doi.mjs`: gói ngữ cảnh + worker;
  - `chinh.mjs`: cấu hình, nhật ký, bảo trì, `--kiem-tra`, các vòng lặp.
- Lõi sổ `so-core.mjs` / `books.mjs` / `drive.mjs` dùng nguyên.
- Rút gọn theo ponytail: không tầng trừu tượng thừa; mỗi chỗ cố ý làm đơn giản đánh dấu bằng `ponytail:`.

**Tech Stack:** Node ≥ 22.13 (`node:sqlite`, `node:test`), `@modelcontextprotocol/sdk` 1.32 (Streamable HTTP server/client), `zod` 4, Claude Code CLI (`claude -p --bare`), launchd.

**Spec:** `docs/superpowers/specs/2026-10-08-np-dichvu-design.md` (nghiệp vụ: `docs/superpowers/specs/2026-10-07-np-bot-zalo-design.md`)

```js
// spike-claude.mjs — đặt ở gốc repo NP (cần node_modules của repo), chạy: node spike-claude.mjs <ảnh> <claude>
import http from "node:http";
import { spawnSync } from "node:child_process";
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StreamableHTTPServerTransport } from "@modelcontextprotocol/sdk/server/streamableHttp.js";
import { z } from "zod";
const ANH = process.argv[2], CLAUDE = process.argv[3] || "claude", TOKEN = "spike-" + Math.random().toString(36).slice(2);
const goi = [];
const may = http.createServer(async (req, res) => {
  if (req.headers.authorization !== `Bearer ${TOKEN}`) { res.writeHead(401).end(); return }
  if (req.method !== "POST") { res.writeHead(405).end(); return }
  const s = new McpServer({ name: "np", version: "1" });
  s.registerTool("ghi_so_tien", { description: "Ghi số tiền giao dịch (VND, số nguyên) đọc được trên ảnh.", inputSchema: { so_tien: z.number().int() } },
    async ({ so_tien }) => { goi.push(so_tien); return { content: [{ type: "text", text: "đã ghi" }] } });
  const t = new StreamableHTTPServerTransport({ sessionIdGenerator: undefined, enableJsonResponse: true });
  res.on("close", () => { t.close(); s.close() });
  await s.connect(t);
  let body = ""; for await (const c of req) body += c;
  await t.handleRequest(req, res, body ? JSON.parse(body) : undefined);
});
await new Promise((r) => may.listen(0, "127.0.0.1", r));
const thuMuc = ANH.slice(0, ANH.lastIndexOf("/"));
const mcp = JSON.stringify({ mcpServers: { np: { type: "http", url: `http://127.0.0.1:${may.address().port}/mcp`, headers: { Authorization: `Bearer ${TOKEN}` } } } });
const r = spawnSync(CLAUDE, ["-p", `Đọc ảnh ${ANH}. Tìm số tiền giao dịch trên biên lai rồi gọi công cụ ghi_so_tien với số đó (VND, số nguyên).`,
  "--model", "claude-sonnet-5-5", "--tools", "Read", "--add-dir", thuMuc,
  "--allowedTools", `mcp__np__ghi_so_tien,Read(/${thuMuc}/**)`, "--mcp-config", mcp, "--strict-mcp-config",
  "--permission-mode", "dontAsk", "--max-turns", "6", "--output-format", "json", "--no-session-persistence"],
  { encoding: "utf8", timeout: 240_000 });
console.log(JSON.stringify({ ma_thoat: r.status, goi, ket_qua: (() => { try { const j = JSON.parse(r.stdout.trim().split("\n").pop()); return { subtype: j.subtype, is_error: j.is_error, num_turns: j.num_turns } } catch { return r.stdout.slice(0, 300) } })(), loi: r.stderr.slice(0, 300) }, null, 1));
may.close();
```

export function moCsdl(file) {

/** Chạy fn trong một giao dịch (BEGIN IMMEDIATE: giữ quyền ghi ngay). KHÔNG lồng nhau. */
export function giaoDich(db, fn) {
  db.exec("BEGIN IMMEDIATE");
  try { const kq = fn(); db.exec("COMMIT"); return kq }
  catch (e) { try { db.exec("ROLLBACK") } catch {} throw e }
}

// bill NGAY (link Zalo có thể hết hạn trước khi khoản được ghi). Nhóm chưa khai ghi mốc `nhom_la:<chat_id>` (xem --kiem-tra).
import { writeFileSync, mkdirSync } from "node:fs";
import { join } from "node:path";
import { datMoc } from "./csdl.mjs";

export function taoAdapterZalo({ db, kho, nhom, nhomVanHanh = null, thuMucAnh, taiFn = fetch, nhatKy = () => {}, now = Date.now }) {
  const hopLe = (ch) => Object.prototype.hasOwnProperty.call(nhom, ch);
  async function taiAnh(t) {
    let tt = "da_tai", tep = null;
    try {
      const r = await taiFn(t.anh_url, { signal: AbortSignal.timeout(20_000) });
      const loai = String(r.headers?.get?.("content-type") || "");
      if (!r.ok) tt = `loi:HTTP ${r.status}`;
      else if (loai && !loai.startsWith("image/")) tt = `loi:Zalo trả ${loai.split(";")[0]}, không phải ảnh`;
      else {
        const buf = Buffer.from(await r.arrayBuffer());
        if (buf.length > 20_000_000) tt = "loi:ảnh quá 20MB";
        else { mkdirSync(thuMucAnh, { recursive: true }); tep = join(thuMucAnh, `${t.id}.jpg`); writeFileSync(tep, buf) }
      }
    } catch (e) { tt = `loi:${e?.name === "TimeoutError" ? "hết giờ chờ" : "lỗi mạng"}` }
    kho.datAnh(t.id, { tep, tt });
    nhatKy({ tin: t.id, anh: tt === "da_tai" ? "da_tai" : "loi" });
  }
  async function nhan(u) {
    const z = u?.message, ch = String(z?.chat?.id ?? "");
    if (!z || !/^message\.(text|image)\.received$/.test(String(u?.event_name))) return nhatKy({ bo: u?.event_name || "khong_ro" });
    if (z.chat?.chat_type !== "GROUP") return nhatKy({ bo: "tin_rieng" });
    if (nhomVanHanh && ch === String(nhomVanHanh)) return nhatKy({ bo: "nhom_van_hanh" });
    if (!hopLe(ch)) {
      datMoc(db, `nhom_la:${ch}`, now(), JSON.stringify({ nguoi: z.from?.display_name || "", vi_du: String(z.text ?? z.caption ?? "").slice(0, 40) }));
      return nhatKy({ bo: "nhom_chua_khai", chat_id: ch });
    }
    const t = kho.nhanTin(u, { now: now() });
    if (!t) return;
    nhatKy({ nhan: t.id, chat_id: ch, ma: t.ma });
    if (t.anh_tt === "cho_tai") await taiAnh(t);
  }
  return { nhan, taiAnhConDo: async () => { for (const t of kho.tinChoTaiAnh()) await taiAnh(t) } };

export function phanLoai({ maThoat, stdout, stderr, quaGio }) {
  if (quaGio) return { ok: false, kieu: "qua_gio" };
  let j = null;

export function chayClaude({ chuongTrinh = "claude", args, env, cwd, thoiGianMs, signal = null }) {

    const giet = () => { try { process.kill(-p.pid, "SIGKILL") } catch {} };
    const khiHuy = () => { huy = true; giet() };
    const hen = setTimeout(() => { quaGio = true; giet() }, thoiGianMs);
    signal?.addEventListener("abort", khiHuy, { once: true });
    const ket = (kq) => { clearTimeout(hen); signal?.removeEventListener("abort", khiHuy); xong(kq) };
    p.stdout.on("data", (c) => { out += c });
    p.stderr.on("data", (c) => { err += c });
    p.on("error", (e) => ket({ ok: false, kieu: "loi", loi: `không chạy được ${chuongTrinh}: ${e.code || e.message}` }));
    p.on("close", (ma) => ket(huy ? { ok: false, kieu: "huy" } : { ...phanLoai({ maThoat: ma, stdout: out, stderr: err, quaGio }), maThoat: ma }));
  });
}

/** File KEY=VALUE (bỏ dòng trống / '#'). Thiếu khoá bắt buộc → lỗi nêu tên khoá, không in giá trị. */
export function docBiMat(file, batBuoc = ["ZALO_BOT_TOKEN", "CLAUDE_CODE_OAUTH_TOKEN"]) {
  const ra = {};
  for (const dong of readFileSync(file, "utf8").split("\n")) {
    const m = /^\s*([A-Z0-9_]+)\s*=\s*(.*?)\s*$/.exec(dong);
    if (m) ra[m[1]] = m[2];
  }
  const thieu = batBuoc.filter((k) => !ra[k]);
  if (thieu.length) throw new Error(`thiếu ${thieu.join(", ")} trong ${file}`);
  return ra;
}

export const che = (s) => String(s).replace(/(Bearer\s+)[\w.~+/-]+=*/gi, "$1***").replace(/\b(token|key|secret)=[^&\s"]+/gi, "$1=***").replace(/sk-ant-[\w-]+/g, "sk-ant-***");

export function taoBaoTri({ db, thuMuc, now = Date.now, giuSaoLuu = 14, giuNgay = 30 }) {
  return {
    /** Mỗi ngày một bản sao nhất quán (VACUUM INTO), giữ `giuSaoLuu` bản mới nhất. Đã có bản hôm nay → null. */
    saoLuu() {
      const file = join(thuMuc.saoLuu, `np-${ngayVN(now())}.sqlite`);
      if (existsSync(file)) return null;
      mkdirSync(thuMuc.saoLuu, { recursive: true });
      db.exec(`VACUUM INTO '${file.replaceAll("'", "''")}'`);
      const ds = readdirSync(thuMuc.saoLuu).filter((f) => /^np-\d{4}-\d{2}-\d{2}\.sqlite$/.test(f)).sort();
      for (const f of ds.slice(0, -giuSaoLuu)) rmSync(join(thuMuc.saoLuu, f));
      return file;
    },
    /** Ảnh tạm và nhật ký quá `giuNgay` ngày (chứng từ chính thức đã ở Drive). Trả số tệp đã xoá. */

        let ds = [];
        try { ds = readdirSync(d) } catch { continue }
        for (const f of ds) { const p = join(d, f); try { if (now() - statSync(p).mtimeMs > giuNgay * 86400e3) { rmSync(p); xoa++ } } catch {} }
      }
      return xoa;
    },
  };

export function tomTat(db, now = Date.now()) {

  return {
    viec: dem("viec"), gui: dem("gui"),
    viec_qua_han_phut: cu != null && cu < now ? phut(cu) : 0,
